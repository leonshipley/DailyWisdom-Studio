import json
import time
from pathlib import Path
from pydantic import BaseModel, Field
from google import genai
from google.genai import types

class BookClip(BaseModel):
    title: str = Field(description="Catchy, high-CTR YouTube Short title under 60 chars")
    start_time: str = Field(description="Start timestamp HH:MM:SS or MM:SS")
    end_time: str = Field(description="End timestamp HH:MM:SS or MM:SS")
    book_title: str = Field(description="The exact title of the book discussed in this clip")
    book_author: str = Field(description="The author of the book (or 'Unknown')")
    hook_caption: str = Field(description="1 punchy sentence summarizing the insight")
    hashtags: list[str] = Field(description="5 viral tags, e.g., ['#Wisdom', '#Books', '#Shorts']")

class AnalysisResponse(BaseModel):
    clips: list[BookClip]

def analyze_video_for_books(client: genai.Client, video_path: Path, max_clips: int = 2) -> list[BookClip]:
    print("\nðŸ§  [Brain] Uploading video to Gemini File API...")
    video_file = client.files.upload(file=str(video_path))

    while video_file.state.name == "PROCESSING":
        time.sleep(3)
        video_file = client.files.get(name=video_file.name)

    if video_file.state.name == "FAILED":
        raise RuntimeError("Gemini failed to process the video.")

    print("ðŸ§  [Brain] Identifying book recommendations and high-retention moments...")
    prompt = f"""
    You are an elite short-form video editor for a book wisdom channel.
    Analyze this video and extract up to {max_clips} viral moments where a specific book, philosophy, or mental model is explained.

    Requirements:
    - Clip length: strictly 30 to 58 seconds.
    - Identify the EXACT book title and author being discussed.
    - Start right at the punchy hook; stop cleanly after the lesson is complete.
    """

    response = client.models.generate_content(
        model="gemini-2.0-flash",
        contents=[video_file, prompt],
        config=types.GenerateContentConfig(
            response_mime_type="application/json",
            response_schema=AnalysisResponse,
            temperature=0.2,
        ),
    )

    try:
        client.files.delete(name=video_file.name)
    except Exception:
        pass

    data = json.loads(response.text)
    return [BookClip(**item) for item in data.get("clips", [])]

_BIBLE_MEM_CACHE: Optional[List[Dict[str, Any]]] = None

def ensure_full_bible_downloaded() -> List[Dict[str, Any]]:
    """Downloads and caches the complete 66-book, 31,102-verse English KJV Bible JSON."""
    global _BIBLE_MEM_CACHE
    if _BIBLE_MEM_CACHE is not None:
        return _BIBLE_MEM_CACHE

    if not KJV_JSON_FILE.exists() or KJV_JSON_FILE.stat().st_size < 1_000_000:
        url = "https://raw.githubusercontent.com/thiagobodruk/bible/master/json/en_kjv.json"
        try:
            req = urllib.request.Request(url, headers={"User-Agent": "Mozilla/5.0"})
            with urllib.request.urlopen(req, timeout=20) as resp:
                raw_bytes = resp.read()
                # Strip UTF-8 BOM if present
                text_data = raw_bytes.decode("utf-8-sig")
                KJV_JSON_FILE.write_text(text_data, encoding="utf-8")
        except Exception:
            pass

    if KJV_JSON_FILE.exists() and KJV_JSON_FILE.stat().st_size > 100_000:
        try:
            _BIBLE_MEM_CACHE = json.loads(KJV_JSON_FILE.read_text(encoding="utf-8-sig"))
            return _BIBLE_MEM_CACHE
        except Exception:
            pass
    return []

def get_book_chapters_count(book_eng_name: str) -> int:
    for b_en, _, ch_cnt in BIBLE_66_BOOKS:
        if b_en.strip().lower() == book_eng_name.strip().lower():
            return ch_cnt
    return 1

def get_verses_for_chapter(book_eng_name: str, chapter_num: int) -> Dict[int, str]:
    """Returns {verse_number: verse_text} for any of the 66 books and any chapter."""
    data = ensure_full_bible_downloaded()
    book_idx = next((i for i, (b_en, _, _) in enumerate(BIBLE_66_BOOKS) if b_en.strip().lower() == book_eng_name.strip().lower()), -1)

    if data and 0 <= book_idx < len(data):
        chapters = data[book_idx].get("chapters", [])
        if 1 <= chapter_num <= len(chapters):
            v_list = chapters[chapter_num - 1]
            return {i + 1: re.sub(r"\{([^}]*)\}", r"\1", v_str) for i, v_str in enumerate(v_list)}

    # Online single-chapter fallback if JSON hasn't downloaded yet
    try:
        q = urllib.parse.quote(f"{book_eng_name} {chapter_num}")
        req = urllib.request.Request(f"https://bible-api.com/{q}?translation=kjv", headers={"User-Agent": "Mozilla/5.0"})
        with urllib.request.urlopen(req, timeout=8) as resp:
            j = json.loads(resp.read().decode("utf-8"))
            return {int(v["verse"]): v["text"].strip() for v in j.get("verses", [])}
    except Exception:
        return {1: "The Lord is near to all who call upon Him in truth."}

def translate_verse_to_hindi(ref: str, english_verse: str) -> Tuple[str, str]:
    """
    Guarantees pure Devanagari Hindi translation for any Bible verse.
    Never caches or returns English in the Devanagari slot.
    """
    ref_clean = ref.strip()
    if ref_clean in VERIFIED_PAVITRA_BAIBIL:
        return VERIFIED_PAVITRA_BAIBIL[ref_clean]

    cache: Dict[str, Any] = {}
    if HINDI_CACHE_FILE.exists():
        try:
            cache = json.loads(HINDI_CACHE_FILE.read_text(encoding="utf-8"))
        except Exception:
            cache = {}

    if ref_clean in cache and isinstance(cache[ref_clean], dict):
        cached_dev = cache[ref_clean].get("devanagari", "")
        if re.search(r"[\u0900-\u097F]", cached_dev):
            return cached_dev, cache[ref_clean].get("hinglish", cached_dev)

    devanagari = ""
    hinglish = ""
    try:
        q = urllib.parse.quote(english_verse.strip())
        # Use single dt=t first for 100% reliable Devanagari extraction
        url = f"https://translate.googleapis.com/translate_a/single?client=gtx&sl=en&tl=hi&dt=t&q={q}"
        req = urllib.request.Request(url, headers={"User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64)"})
        with urllib.request.urlopen(req, timeout=8) as resp:
            arr = json.loads(resp.read().decode("utf-8"))
            if arr and arr[0]:
                devanagari = "".join(str(seg[0]) for seg in arr[0] if seg and seg[0]).strip()
    except Exception:
        pass

    if re.search(r"[\u0900-\u097F]", devanagari):
        hinglish = devanagari
        cache[ref_clean] = {"devanagari": devanagari, "hinglish": hinglish}
        try:
            HINDI_CACHE_FILE.write_text(json.dumps(cache, ensure_ascii=False, indent=2), encoding="utf-8")
        except Exception:
            pass
        return devanagari, hinglish

    # Fallback if offline
    return "à¤¯à¤¹à¥‹à¤µà¤¾ à¤…à¤ªà¤¨à¥‡ à¤¸à¤­à¥€ à¤ªà¥à¤•à¤¾à¤°à¤¨à¥‡ à¤µà¤¾à¤²à¥‹à¤‚ à¤•à¥‡ à¤¸à¤®à¥€à¤ª à¤°à¤¹à¤¤à¤¾ à¤¹à¥ˆ à¤”à¤° à¤‰à¤¨à¥à¤¹à¥‡à¤‚ à¤¶à¤¾à¤¨à¥à¤¤à¤¿ à¤¦à¥‡à¤¤à¤¾ à¤¹à¥ˆà¥¤", "Yahowa apne sabhi pukarne walon ke samip rehta hai."



def lookup_reference_bilingual(ref_str: str) -> Tuple[str, str, str, str]:
    """
    Parses any reference like 'Philippians 4:19', 'Psalm 34:18', or 'Philippians 4:6-7'
    and returns (normalized_ref, english_text, hindi_devanagari, romanized_hinglish).
    """
    ref_clean = ref_str.strip()
    m = re.match(r"^((?:[1-3]\s+)?[A-Za-z\s]+?)\s+(\d+):(\d+)(?:-(\d+))?$", ref_clean)
    if not m:
        en_default = "And my God will supply every need of yours according to his riches in glory in Christ Jesus."
        dev, rom = translate_verse_to_hindi("Philippians 4:19", en_default)
        return ("Philippians 4:19", en_default, dev, rom)

    book_raw = m.group(1).strip()
    if book_raw.lower() == "psalm":
        book_lookup = "Psalms"
    else:
        book_lookup = book_raw

    ch_num = int(m.group(2))
    v_start = int(m.group(3))
    v_end = int(m.group(4)) if m.group(4) else v_start

    verses_map = get_verses_for_chapter(book_lookup, ch_num)
    collected = []

def format_reference_for_speech(ref_str: str, is_hindi_speech: bool = False) -> str:
    """
    Converts 'Leviticus 1:1' into:
    - Hindi Speech:   'à¤²à¥ˆà¤µà¥à¤¯à¤µà¥à¤¯à¤µà¤¸à¥à¤¥à¤¾, à¤…à¤§à¥à¤¯à¤¾à¤¯ 1, à¤µà¤šà¤¨ 1' (Never says '1 bajke 1 minute'!)
    - English Speech: 'Leviticus, chapter 1, verse 1'
    """
    ref_clean = ref_str.strip()
    m = re.match(r"^((?:[1-3]\s+)?[A-Za-z\s]+?)\s+(\d+):(\d+)(?:-(\d+))?$", ref_clean)
    if not m:
        # Strip any colons between numbers so TTS never reads clock time
        if is_hindi_speech:
            return re.sub(r"(\d+):(\d+)", r"à¤…à¤§à¥à¤¯à¤¾à¤¯ \1, à¤µà¤šà¤¨ \2", ref_clean)
        return re.sub(r"(\d+):(\d+)", r"chapter \1, verse \2", ref_clean)

    book_en = m.group(1).strip()
    ch = m.group(2)
    v1 = m.group(3)
    v2 = m.group(4)

    if is_hindi_speech:
        hi_book = HINDI_BOOK_SPOKEN_MAP.get(book_en.lower(), book_en)
        if v2 and v2 != v1:
            return f"{hi_book}, à¤…à¤§à¥à¤¯à¤¾à¤¯ {ch}, à¤µà¤šà¤¨ {v1} à¤¸à¥‡ {v2}"
        return f"{hi_book}, à¤…à¤§à¥à¤¯à¤¾à¤¯ {ch}, à¤µà¤šà¤¨ {v1}"
    else:
        if v2 and v2 != v1:
            return f"{book_en}, chapter {ch}, verses {v1} to {v2}"
        return f"{book_en}, chapter {ch}, verse {v1}"


def translate_any_text_to_hindi(en_text: str) -> str:
    """Translates any English sentence (topic, meaning, prayer) into 100% pure Devanagari Hindi."""
    if not en_text or not en_text.strip():
        return ""
    # If already contains Devanagari characters, return as-is
    if re.search(r"[\u0900-\u097F]", en_text):
        return en_text
    cache: Dict[str, Any] = {}
    if HINDI_CACHE_FILE.exists():
        try:
            cache = json.loads(HINDI_CACHE_FILE.read_text(encoding="utf-8"))
        except Exception:
            cache = {}
    key = f"txt_{en_text.strip()[:80]}"
    if key in cache and isinstance(cache[key], str) and cache[key]:
        return cache[key]
    try:
        q = urllib.parse.quote(en_text.strip())
        url = f"https://translate.googleapis.com/translate_a/single?client=gtx&sl=en&tl=hi&dt=t&q={q}"
        req = urllib.request.Request(url, headers={"User-Agent": "Mozilla/5.0"})
        with urllib.request.urlopen(req, timeout=6) as resp:
            arr = json.loads(resp.read().decode("utf-8"))
            if arr and arr[0]:
                hi_out = "".join(seg[0] for seg in arr[0] if seg and seg[0]).strip()
                if hi_out:
                    cache[key] = hi_out
                    HINDI_CACHE_FILE.write_text(json.dumps(cache, ensure_ascii=False, indent=2), encoding="utf-8")
                    return hi_out
    except Exception:
        pass
    return en_text

def timestamp_to_seconds(ts: str) -> float:
    parts = [float(p) for p in ts.strip().split(":")]
    if len(parts) == 3:
        return parts[0] * 3600 + parts[1] * 60 + parts[2]
    elif len(parts) == 2:
        return parts[0] * 60 + parts[1]
    return parts[0]

def track_quota():
    today_str = date.today().isoformat()
    quota_data = {"date": today_str, "used": 0}

    if QUOTA_FILE.exists():
        try:
            with open(QUOTA_FILE, "r") as f:
                saved = json.load(f)
            if saved.get("date") == today_str:
                quota_data = saved
        except Exception:
            pass

    quota_data["used"] += 1
    remaining = max(0, 1500 - quota_data["used"])

    with open(QUOTA_FILE, "w") as f:
        json.dump(quota_data, f, indent=2)

    return quota_data["used"], remaining

def render_full_screen(input_file: Path, clips: list[VideoClip], video_title: str):
    w, h = 1080, 1920
    target_folder = CLIPS_DIR / NICHE_FOLDER_NAME / clean_name(video_title)
    target_folder.mkdir(parents=True, exist_ok=True)

    print(f"\n[4/4] Output Directory: {target_folder.resolve()}")
    print(f" Rendering {len(clips)} Full-Screen Ultra-HD Shorts...")

    for i, clip in enumerate(clips):
        clip_name = clean_name(clip.title)
        out_file = target_folder / f"short_{i+1}_{clip_name}.mp4"

        start_sec = timestamp_to_seconds(clip.start_time)
        end_sec = timestamp_to_seconds(clip.end_time)
        duration = max(1.0, end_sec - start_sec)

        print(f"\nâš¡ Rendering Full-Screen Short #{i+1}: '{clip.title}' ({round(duration)}s)")

        # Filter Breakdown with PTS reset so progress bar starts accurately at t=0
        filter_complex = (
            f"[0:v]setpts=PTS-STARTPTS,scale=-2:{h}:flags=lanczos,crop={w}:{h}:(in_w-{w})/2:0,"
            f"unsharp=5:5:0.85:5:5:0.0,eq=contrast=1.05:saturation=1.08[v_base];"
            f"[v_base]drawbox=x=0:y=1902:w='min({w}, {w}*t/{duration})':h=14:color=0xFFCC00@1:t=fill[vout];"
            f"[0:a]asetpts=PTS-STARTPTS,aformat=sample_fmts=fltp:sample_rates=44100:channel_layouts=stereo[speech];"
            f"[1:a]volume=0.06,aformat=sample_fmts=fltp:sample_rates=44100:channel_layouts=stereo[bgm];"
            f"[speech][bgm]amix=inputs=2:duration=first:dropout_transition=2[aout]"
        )

        cmd = [
            "ffmpeg", "-y",
            "-ss", str(start_sec),
            "-t", str(duration),
            "-i", str(input_file),
            "-stream_loop", "-1",
            "-i", str(BGM_FILE),
            "-filter_complex", filter_complex,
            "-map", "[vout]",
            "-map", "[aout]",
            "-c:v", "libx264",
            "-preset", "veryfast",
            "-crf", "19",
            "-c:a", "aac",
            "-b:a", "192k",
            str(out_file)
        ]

        result = subprocess.run(cmd, capture_output=True, text=True)
        if result.returncode != 0:
            print(f"[FFmpeg Error on clip {i+1}]: {result.stderr[-400:]}")
            continue

        print(f" Saved: {out_file.name}")

    return target_folder

def mute_any_video(input_vid: Path) -> Path:
    """Strips audio and normalizes video stream so FFmpeg concat never fails with -22."""
    if not input_vid or not input_vid.exists():
        return input_vid
    muted_dir = INTRO_DIR / "muted_cache"
    muted_dir.mkdir(parents=True, exist_ok=True)
    safe_stem = re.sub(r"[^a-zA-Z0-9_]+", "_", input_vid.stem).strip("_")[:30]
    muted_out = muted_dir / f"muted_{safe_stem}.mp4"
    if not muted_out.exists() or muted_out.stat().st_size < 1000:
        cmd = [
            "ffmpeg", "-y", "-i", str(input_vid),
            "-t", "12", "-an",
            "-vf", "scale=1080:1920:force_original_aspect_ratio=increase,crop=1080:1920,fps=24,format=yuv420p,setsar=1",
            "-c:v", "libx264", "-preset", "ultrafast", "-crf", "20",
            str(muted_out)
        ]
        subprocess.run(cmd, stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
    return muted_out if muted_out.exists() else input_vid

def load_channels() -> dict:
    channels = {}
    if CHANNELS_CONFIG.exists():
        try:
            with open(CHANNELS_CONFIG, "r") as f:
                channels = json.load(f)
        except Exception:
            channels = {}

    legacy_token = BASE_DIR / "youtube_token.pickle"
    if legacy_token.exists():
        ch_name = "TheDailyWisdom_Hub"
        token_path = CHANNELS_DIR / f"{ch_name}.pickle"
        if not token_path.exists():
            shutil.copy(legacy_token, token_path)
        if ch_name not in channels:
            channels[ch_name] = str(token_path)
            save_channels(channels)

    for p in CHANNELS_DIR.glob("*.pickle"):
        name = p.stem.replace("_", " ")
        if name not in channels:
            channels[name] = str(p)
            save_channels(channels)

    return channels

def save_channels(data: dict):
    with open(CHANNELS_CONFIG, "w") as f:
        json.dump(data, f, indent=2)

def connect_new_channel():
    if not CLIENT_SECRET_FILE.exists():
        print("âŒ Missing client_secret.json in project root directory!")
        return None

    print("\nðŸŒ Opening Google login in your web browser...")
    flow = InstalledAppFlow.from_client_secrets_file(str(CLIENT_SECRET_FILE), YOUTUBE_SCOPES)
    credentials = flow.run_local_server(port=0, prompt="select_account")
    youtube = build("youtube", "v3", credentials=credentials)
    res = youtube.channels().list(mine=True, part="snippet").execute()

    if not res.get("items"):
        return None

    channel_name = res["items"][0]["snippet"]["title"]
    safe_name = re.sub(r'[\/*?:"<>| ]', "_", channel_name)
    token_file = CHANNELS_DIR / f"{safe_name}.pickle"

    with open(token_file, "wb") as f:
        pickle.dump(credentials, f)

    channels = load_channels()
    channels[channel_name] = str(token_file)
    save_channels(channels)

def get_authenticated_youtube(channel_name: str):
    channels = load_channels()
    token_path = channels.get(channel_name)
    if not token_path or not Path(token_path).exists():
        return None

    with open(token_path, "rb") as f:
        credentials = pickle.load(f)

    if not credentials or not credentials.valid:
        if credentials and credentials.expired and credentials.refresh_token:
            credentials.refresh(Request())
            with open(token_path, "wb") as f:
                pickle.dump(credentials, f)
        else:
            return None

    return build("youtube", "v3", credentials=credentials)

def clean_filename(text: str) -> str:
    clean = re.sub(r"[^a-zA-Z0-9]+", "_", text or "")
    return clean.strip("_")[:35] or "media_item"

def timestamp_to_seconds(ts: str) -> float:
    parts = [float(p) for p in ts.strip().split(":")]
    if len(parts) == 3:
        return parts[0] * 3600 + parts[1] * 60 + parts[2]
    elif len(parts) == 2:
        return parts[0] * 60 + parts[1]
    return parts[0]

_CHANNEL_STATS_CACHE = {}

def get_my_top_shorts(channel_name: str, max_results: int = 8) -> list[dict]:
    """Fetches the user's own uploaded videos from YouTube API and sorts by most viewed."""
    if channel_name in _CHANNEL_STATS_CACHE:
        return _CHANNEL_STATS_CACHE[channel_name]

    youtube = get_authenticated_youtube(channel_name)
    if not youtube:
        return []

    try:
        ch_res = youtube.channels().list(mine=True, part="contentDetails").execute()
        if not ch_res.get("items"):
            return []
        uploads_id = ch_res["items"][0]["contentDetails"]["relatedPlaylists"]["uploads"]

        pl_res = youtube.playlistItems().list(
            playlistId=uploads_id, part="contentDetails", maxResults=max_results
        ).execute()
        vid_ids = [item["contentDetails"]["videoId"] for item in pl_res.get("items", [])]
        if not vid_ids:
            return []

        v_res = youtube.videos().list(
            id=",".join(vid_ids), part="snippet,statistics"
        ).execute()

        my_videos = []
        for v in v_res.get("items", []):
            stats = v.get("statistics", {})
            snip = v.get("snippet", {})
            my_videos.append({
                "id": v["id"],
                "title": snip.get("title", ""),
                "views": int(stats.get("viewCount", 0)),
                "likes": int(stats.get("likeCount", 0)),
                "comments": int(stats.get("commentCount", 0)),
                "tags": snip.get("tags", [])
            })

        my_videos.sort(key=lambda x: x["views"], reverse=True)
        _CHANNEL_STATS_CACHE[channel_name] = my_videos
        return my_videos
    except Exception:
        return []


def scan_competitor_shorts(topic: str, limit: int = 6) -> list[dict]:
    """Scans YouTube for top-viewed competitor Shorts on this exact topic."""
    ydl_opts = {"quiet": True, "extract_flat": True, "skip_download": True}
    competitors = []
    query = f"ytsearch10:{topic} #shorts"
    try:
        with yt_dlp.YoutubeDL(ydl_opts) as ydl:
            res = ydl.extract_info(query, download=False)
            for entry in res.get("entries", []):
                if not entry:
                    continue
                views = entry.get("view_count") or 0
                title = entry.get("title") or ""
                if title:
                    competitors.append({
                        "title": title,
                        "views": int(views),
                        "url": f"https://www.youtube.com/watch?v={entry.get('id', '')}"
                    })
    except Exception:
        pass

    competitors.sort(key=lambda x: x["views"], reverse=True)
    return competitors[:limit]

def load_clip_metadata(video_path: Path) -> dict:
    meta_path = video_path.with_suffix(".json")
    if meta_path.exists():
        try:
            with open(meta_path, "r", encoding="utf-8") as f:
                return json.load(f)
        except Exception:
            pass
    return {"title": video_path.stem.replace("_", " "), "description": "Watch this message.\n\n#Shorts #Faith", "tags": ["Shorts", "Faith"], "youtube_id": None}

def upload_video_to_youtube(channel_name: str, video_path: Path) -> str | None:
    youtube = get_authenticated_youtube(channel_name)
    if not youtube:
        print(f"âŒ Could not authenticate channel: {channel_name}")

    meta = load_clip_metadata(video_path)
    print(f"\nðŸš€ [YouTube] Publishing '{meta['title']}' to '{channel_name}'...")

    body = {
        "snippet": {"title": meta["title"][:100], "description": meta["description"], "tags": meta["tags"], "categoryId": "27"},
        "status": {"privacyStatus": "public", "selfDeclaredMadeForKids": False}
    }

    req = youtube.videos().insert(part="snippet,status", body=body, media_body=media)

    res = None
    try:
        while res is None:
            status, res = req.next_chunk()
            if status:
                print(f"   Upload progress: {int(status.progress() * 100)}%")
        yt_id = res.get("id")

        meta["youtube_id"] = yt_id
        meta_path = video_path.with_suffix(".json")
        with open(meta_path, "w", encoding="utf-8") as f:
            json.dump(meta, f, indent=2)

        print(f"ðŸŽ‰ Live on YouTube! https://youtube.com/shorts/{yt_id}")
        return yt_id
    except Exception as e:
        if "uploadLimitExceeded" in str(e):
            print("âš ï¸ YouTube daily upload limit reached for today. Video is safely stored in 'clips/' folder.")
        else:
            print(f"âŒ Upload error: {e}")
        return None

def generate_voiceover(text: str, output_voice_path: Path):
    """Generates audio AND millisecond-synced word timings using the selected Voice Studio profile (Hindi/Indian/UK/US)."""
    from voice_studio import load_active_voice
    active_v = load_active_voice()
    v_name = active_v.get("voice", "hi-IN-MadhurNeural")
    v_rate = active_v.get("rate", "-4%")
    v_pitch = active_v.get("pitch", "+0Hz")
    print(f"   ðŸŽ™ï¸ Speaking with Voice: {active_v.get('name')} [{v_name} | Rate: {v_rate} | Pitch: {v_pitch}]")

    # Ensure no raw number:number colons reach TTS as clock time
    if "hi-IN" in v_name and "Hindi" in active_v.get("script_mode", ""):
        text = re.sub(r"(\d+):(\d+)", r"à¤…à¤§à¥à¤¯à¤¾à¤¯ \1 à¤µà¤šà¤¨ \2", text)
    else:
        text = re.sub(r"(\d+):(\d+)", r"chapter \1 verse \2", text)
    json_path = Path(str(output_voice_path).replace(".mp3", ".json"))

    async def _make():
        try:
            communicate = edge_tts.Communicate(text, voice=v_name, rate=v_rate, pitch=v_pitch, boundary="WordBoundary")
        except TypeError:
            try:
                communicate = edge_tts.Communicate(text, voice=v_name, rate=v_rate, boundary="WordBoundary")
            except TypeError:
                communicate = edge_tts.Communicate(text, voice=v_name, rate=v_rate)

        raw_events = []
        with open(output_voice_path, "wb") as f_voice:
            async for chunk in communicate.stream():
                if chunk["type"] == "audio":
                    f_voice.write(chunk["data"])
                elif chunk["type"] in ("WordBoundary", "SentenceBoundary"):
                    start_s = chunk["offset"] / 10_000_000.0
                    dur_s = chunk["duration"] / 10_000_000.0
                    raw_events.append({
                        "type": chunk["type"],
                        "text": chunk["text"],
                        "start": start_s,
                        "end": start_s + dur_s
                    })

        word_timings = []
        for ev in raw_events:
            if ev["type"] == "WordBoundary" or len(ev["text"].split()) == 1:
                word_timings.append({
                    "text": ev["text"].strip(),
                    "start": round(ev["start"], 3),
                    "end": round(ev["end"], 3)
                })
            else:
                words = ev["text"].split()
                if not words:
                    continue
                weights = []
                for w in words:
                    w_weight = max(2, len(w))
                    if re.search(r"[.,;!?]", w):
                        w_weight += 4
                    weights.append(w_weight)
                total_w = sum(weights) or 1
                cur_t = ev["start"]
                ev_dur = max(0.1, ev["end"] - ev["start"])
                for w, wt in zip(words, weights):
                    w_dur = ev_dur * (wt / total_w)
                    word_timings.append({
                        "text": w,
                        "start": round(cur_t, 3),
                        "end": round(cur_t + w_dur, 3)
                    })
                    cur_t += w_dur

        merged_timings = []
        i = 0
        while i < len(word_timings):
            cur = word_timings[i]
            c_clean = re.sub(r"[^0-9]", "", cur["text"])
            if i + 1 < len(word_timings):
                nxt = word_timings[i + 1]
                n_clean = re.sub(r"[^0-9\-]", "", nxt["text"])
                if cur["text"].isdigit() and n_clean and nxt["text"][0].isdigit():
                    merged_timings.append({
                        "text": f"{c_clean}:{n_clean}",
                        "start": cur["start"],
                        "end": nxt["end"]
                    })
                    i += 2
                    continue
            merged_timings.append(cur)
            i += 1

        if merged_timings:
            json_path.write_text(json.dumps(merged_timings, indent=2), encoding="utf-8")

    try:
        asyncio.run(_make())
    except RuntimeError:
        loop = asyncio.new_event_loop()
        asyncio.set_event_loop(loop)
        loop.run_until_complete(_make())
        loop.close()

def _sec_to_ass_time(sec: float) -> str:
    sec = max(0.0, float(sec))
    h = int(sec // 3600)
    m = int((sec % 3600) // 60)
    s = sec % 60
    return f"{h}:{m:02d}:{s:05.2f}"

def find_viral_video(min_views: int, queries: list[str]) -> dict | None:
    history = set()
    if HISTORY_FILE.exists():
        try:
            with open(HISTORY_FILE, "r") as f:
                history = set(json.load(f))
        except Exception:
            pass

    print(f"  â€¢ Scanning influencer channels for {min_views:,}+ view videos...")
    ydl_opts = {"quiet": True, "extract_flat": True, "skip_download": True}
    with yt_dlp.YoutubeDL(ydl_opts) as ydl:
        for q in queries:
            try:
                res = ydl.extract_info(q, download=False)
                for entry in res.get("entries", []):
                    if not entry:
                        continue
                    v_id = entry.get("id")
                    dur = entry.get("duration", 0)
                    views = entry.get("view_count") or 0

                    if v_id in history or dur < 180 or dur > 3600:
                        continue

                    if views >= min_views:
                        return {"id": v_id, "url": f"https://www.youtube.com/watch?v={v_id}", "title": entry.get("title"), "views": views}
            except Exception:
                continue
    return None

def get_transcript(video_id: str) -> str | None:
    print("  â€¢ Checking for multi-language captions on YouTube...")
    try:
        api = YouTubeTranscriptApi()
        if hasattr(api, "fetch"):
            f = api.fetch(video_id, languages=["en", "en-US", "en-GB", "te", "hi", "es"])
            data = f.to_raw_data() if hasattr(f, "to_raw_data") else list(f)
        else:
            data = YouTubeTranscriptApi.get_transcript(video_id, languages=["en", "en-US", "en-GB", "te", "hi", "es"])

        formatted = []
        for line in data:
            sec = int(line.get("start", 0))
            m, s = divmod(sec, 60)
            h, m = divmod(m, 60)
            ts = f"{h:02d}:{m:02d}:{s:02d}" if h > 0 else f"{m:02d}:{s:02d}"
            formatted.append(f"[{ts}] {line.get('text', '')}")
        return "\n".join(formatted)
    except Exception:
        return None

def _next_pin_seq(folder: Path) -> int:
    nums = []
    for f in folder.iterdir():
        m = re.match(r"^(\d{3})_", f.name)
        if m:
            nums.append(int(m.group(1)))
    return (max(nums) + 1) if nums else 1

def download_pinterest_media(url: str, sync_to_studio: bool = True) -> Path | None:
    """Reliably downloads any Pinterest Video (.mp4/.m3u8) or HD Image from pin.it / in.pinterest.com / pinterest.com."""
    url = url.strip()
    if not url:
        return None

    headers = {
        "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/124.0.0.0 Safari/537.36",
        "Accept": "text/html,application/xhtml+xml,application/xml;q=0.9,*/*;q=0.8",
        "Accept-Language": "en-US,en;q=0.9"
    }

    print(f"\nðŸ“Œ [Pinterest] Resolving link: {url}...")

    # 1. Resolve short links (pin.it) & regional domains (in.pinterest.com) to canonical Pin URL
    final_url = url
    html = ""
    try:
        req = urllib.request.Request(url, headers=headers)
        with urllib.request.urlopen(req, timeout=12) as resp:
            final_url = resp.geturl()
            html = resp.read().decode("utf-8", errors="ignore")
    except Exception as e:
        print(f"âš ï¸ Initial fetch note: {e}")

    pin_id_match = re.search(r"/pin/(?:[^/]+--)?(\d+)", final_url) or re.search(r"/pin/(?:[^/]+--)?(\d+)", url)
    if pin_id_match:
        pin_id = pin_id_match.group(1)
        canonical_url = f"https://www.pinterest.com/pin/{pin_id}/"
        if canonical_url != final_url or not html:
            try:
                req2 = urllib.request.Request(canonical_url, headers=headers)
                with urllib.request.urlopen(req2, timeout=12) as resp2:
                    html = resp2.read().decode("utf-8", errors="ignore")
                final_url = canonical_url
            except Exception:
                pass
    else:
        pin_id = str(int(time.time()))

    # Unescape Pinterest's JSON slashes (\u002F and \/) so all hidden video & image URLs are exposed
    html_clean = html.replace("\\u002F", "/").replace("\\/", "/").replace("&amp;", "&")

    title_m = re.search(r"<title>([^<]+)</title>", html_clean)
    raw_title = title_m.group(1).replace("Pinterest", "").replace("|", "").strip() if title_m else f"Pin_{pin_id}"
    pin_title = clean_filename(raw_title) or f"Pin_{pin_id}"

    # 2. Check for Direct Pinterest Video Streams (.mp4 or .m3u8) inside the Pin's JSON data
    mp4_urls = re.findall(r"https://v\d*\.pinimg\.com/videos/[^\"'\s\\<>]+\.mp4", html_clean)
    m3u8_urls = re.findall(r"https://v\d*\.pinimg\.com/videos/[^\"'\s\\<>]+\.m3u8", html_clean)

    if mp4_urls or m3u8_urls:
        vid_seq = _next_pin_seq(PIN_VID_DIR)
        final_vid = PIN_VID_DIR / f"{vid_seq:03d}_{pin_title}.mp4"

        # Prefer 720p / expmp4 direct MP4 if available
        if mp4_urls:
            best_mp4 = mp4_urls[0]
            for u in mp4_urls:
                if "720p" in u or "expmp4" in u:
                    best_mp4 = u
                    break
            print(f"ðŸŽ¬ Found Pinterest HD MP4 Video Stream! Downloading...")
            try:
                v_req = urllib.request.Request(best_mp4, headers=headers)
                with urllib.request.urlopen(v_req, timeout=20) as v_resp:
                    final_vid.write_bytes(v_resp.read())
            except Exception as e:
                print(f"âš ï¸ Direct MP4 download note: {e}")

        # If direct MP4 wasn't saved, convert HLS .m3u8 stream via FFmpeg
        if (not final_vid.exists() or final_vid.stat().st_size < 5000) and m3u8_urls:
            print("ðŸŽ¬ Downloading Pinterest HLS Video Stream via FFmpeg...")
            cmd_hls = [
                "ffmpeg", "-y", "-i", m3u8_urls[0],
                "-c:v", "libx264", "-preset", "ultrafast", "-pix_fmt", "yuv420p",
                "-an", str(final_vid)
            ]
            subprocess.run(cmd_hls, stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)

        if final_vid.exists() and final_vid.stat().st_size > 5000:
            print(f"âœ… [Pinterest Video] Saved to: Pinterest/Videos/{final_vid.name}")
            if sync_to_studio:
                studio_copy = INTRO_DIR / final_vid.name
                shutil.copy(final_vid, studio_copy)
                mute_any_video(studio_copy)
                print(f"ðŸ”‡ Auto-muted & added to Studio Intro Videos: {studio_copy.name}")
            return final_vid

    # 3. Try yt-dlp on the canonical URL if regex didn't catch a video
    vid_seq = _next_pin_seq(PIN_VID_DIR)
    temp_pin = PINTEREST_DIR / f"temp_pin_{pin_id}"
    ydl_opts = {
        "outtmpl": str(temp_pin) + ".%(ext)s",
        "format": "bestvideo[ext=mp4]+bestaudio[ext=m4a]/best[ext=mp4]/best",
        "merge_output_format": "mp4",
        "quiet": True,
        "no_warnings": True
    }
    try:
        with yt_dlp.YoutubeDL(ydl_opts) as ydl:
            ydl.extract_info(final_url, download=True)
            downloaded_files = list(PINTEREST_DIR.glob(f"{temp_pin.name}*"))
            if downloaded_files and downloaded_files[0].stat().st_size > 5000:
                final_vid = PIN_VID_DIR / f"{vid_seq:03d}_{pin_title}.mp4"
                shutil.move(str(downloaded_files[0]), str(final_vid))
                print(f"âœ… [Pinterest Video] Saved to: Pinterest/Videos/{final_vid.name}")
                if sync_to_studio:
                    studio_copy = INTRO_DIR / final_vid.name
                    shutil.copy(final_vid, studio_copy)
                    mute_any_video(studio_copy)
                    print(f"ðŸ”‡ Auto-muted & added to Studio Intro Videos: {studio_copy.name}")
                return final_vid
    except Exception:
        for leftover in PINTEREST_DIR.glob(f"{temp_pin.name}*"):
            leftover.unlink(missing_ok=True)

    # 4. Extract Exact Pin High-Res Image (og:image & orig JSON field)
    print("ðŸ–¼ï¸ Extracting High-Res Pinterest Image...")
    img_candidates = []

    # Check og:image meta tag first (always points to the exact Pin's main image, never sidebar ads)
    og_m = re.search(r'<meta[^>]+property=["\']og:image["\'][^>]+content=["\'](https://i\.pinimg\.com/[^"\']+)["\']', html_clean)
    if not og_m:
        og_m = re.search(r'<meta[^>]+content=["\'](https://i\.pinimg\.com/[^"\']+)["\'][^>]+property=["\']og:image["\']', html_clean)
    if og_m:
        og_url = og_m.group(1)
        orig_upgrade = re.sub(r"/\d+x/", "/originals/", og_url)
        img_candidates.extend([orig_upgrade, og_url])

    # Also check "orig":{"width":...,"url":"https://i.pinimg.com/originals/..."}
    orig_matches = re.findall(r'"orig"\s*:\s*\{[^{}]*"url"\s*:\s*"(https://i\.pinimg\.com/originals/[^"]+)"', html_clean)
    img_candidates.extend(orig_matches)

    # Fallback to 736x images
    if not img_candidates:
        img_candidates.extend(re.findall(r"https://i\.pinimg\.com/(?:originals|736x|564x)/[a-zA-Z0-9/_.-]+\.(?:jpg|jpeg|png|webp)", html_clean))

    if not img_candidates:
        print("âŒ Could not find a valid image or video stream on this Pinterest link.")
        return None

    img_seq = _next_pin_seq(PIN_IMG_DIR)
    final_img = PIN_IMG_DIR / f"{img_seq:03d}_{pin_title}.jpg"

    for candidate_u in dict.fromkeys(img_candidates):
        try:
            img_req = urllib.request.Request(candidate_u, headers=headers)
            with urllib.request.urlopen(img_req, timeout=12) as i_resp:
                data = i_resp.read()
                if len(data) > 5000:
                    final_img.write_bytes(data)
                    break
        except Exception:
            continue

    if final_img.exists() and final_img.stat().st_size > 5000:
        print(f"âœ… [Pinterest Image] Saved to: Pinterest/Images/{final_img.name}")
        if sync_to_studio:
            shutil.copy(final_img, BG_DIR / final_img.name)
            shutil.copy(final_img, THUMB_DIR / final_img.name)
            print(f"ðŸŽ¨ Added to Studio Backgrounds & Thumbnails: {final_img.name}")
        return final_img

    print("âŒ Failed to download image bytes from Pinterest.")
    return None

def wrap_and_fit(draw: ImageDraw.ImageDraw, text: str, max_w: int = SAFE_WIDTH, max_lines: int = 3, start_sz: int = 128, min_sz: int = 34, stroke: int = 10) -> Tuple[List[str], ImageFont.FreeTypeFont, int]:
    clean = clean_ascii(text)
    words = clean.split()
    if not words:
        return [""], get_font("extrabold", min_sz), min_sz
    for sz in range(start_sz, min_sz - 1, -2):
        f = get_font("extrabold", sz)
        lines, cur, ok = [], [], True
        for w in words:
            wb = draw.textbbox((0, 0), w, font=f, stroke_width=stroke)
            if (wb[2] - wb[0]) > max_w:
                ok = False
                break
            test = " ".join(cur + [w])
            tb = draw.textbbox((0, 0), test, font=f, stroke_width=stroke)
            if (tb[2] - tb[0]) <= max_w or not cur:
                cur.append(w)
            else:
                lines.append(" ".join(cur))
                cur = [w]
        if cur:
            lines.append(" ".join(cur))
        if ok and len(lines) <= max_lines:
            return lines, f, sz
    f = get_font("extrabold", min_sz)

def get_video_transcript(video_id: str) -> str | None:
    print("âš¡ [Fast-Track] Pulling timestamped transcript (0 MB video download)...")
    data = None

    # 1. Try modern API (v1.0+: .fetch())
    try:
        api = YouTubeTranscriptApi()
        if hasattr(api, "fetch"):
            fetched = api.fetch(video_id, languages=["en", "en-US", "en-GB"])
            if hasattr(fetched, "to_raw_data"):
                data = fetched.to_raw_data()
            elif isinstance(fetched, list):
                data = fetched
            else:
                data = list(fetched)
    except Exception:
        pass

    # 2. Try legacy API (get_transcript)
    if not data:
        try:
            if hasattr(YouTubeTranscriptApi, "get_transcript"):
                data = YouTubeTranscriptApi.get_transcript(video_id, languages=["en", "en-US", "en-GB"])
        except Exception:
            pass

    if not data:
        print("âš ï¸ Could not extract captions for this video.")
        return None

    formatted = []
    for line in data:
        text = line.get("text", "") if isinstance(line, dict) else getattr(line, "text", "")
        start = line.get("start", 0) if isinstance(line, dict) else getattr(line, "start", 0)
        sec = int(start)
        m, s = divmod(sec, 60)
        h, m = divmod(m, 60)
        ts = f"{h:02d}:{m:02d}:{s:02d}" if h > 0 else f"{m:02d}:{s:02d}"
        formatted.append(f"[{ts}] {text}")

    print("âœ… Transcript loaded in under 1 second!")
    return "\n".join(formatted)

def combine_prosody(base_val: str, offset_int: int, unit: str) -> str:
    try:
        num = int(base_val.replace(unit, "").replace("+", "").strip())
    except Exception:
        num = 0
    total = max(-35, min(35, num + int(offset_int)))
    return f"{total:+d}{unit}"

def synthesize_preview_sample(
    voice_id: str,
    rate_str: str = "-4%",
    pitch_str: str = "+0Hz",
    sample_text: str = "",
) -> Path:
    text = (sample_text or "The Lord is near to the brokenhearted, and saves the crushed in spirit. Type Amen.").strip()
    key_hash = hashlib.md5(f"{voice_id}_{rate_str}_{pitch_str}_{text}".encode("utf-8")).hexdigest()[:12]
    out_mp3 = PREVIEWS_DIR / f"preview_{voice_id}_{key_hash}.mp3"

    if out_mp3.exists() and out_mp3.stat().st_size > 1000:
        return out_mp3

    async def _run():
        try:
            comm = edge_tts.Communicate(text, voice=voice_id, rate=rate_str, pitch=pitch_str)
        except TypeError:
            comm = edge_tts.Communicate(text, voice=voice_id, rate=rate_str)
        await comm.save(str(out_mp3))

    try:
        asyncio.run(_run())
    except RuntimeError:
        loop = asyncio.new_event_loop()
        loop.run_until_complete(_run())
        loop.close()

    return out_mp3

def play_audio_on_windows(mp3_path: Path) -> None:
    try:
        subprocess.run(
            ["ffplay", "-nodisp", "-autoexit", "-loglevel", "quiet", str(mp3_path)],
            timeout=18,
            check=True
        )
    except Exception:
        try:
            import os
            os.startfile(str(mp3_path))
        except Exception:
            pass

    hi_devanagari: str,
    hi_hinglish: str,

) -> Path:

    Guarantees that when listen_mode is 'hindi' or 'both', the Hindi verse is spoken
    in 100% pure Devanagari Hindi (never English!).
    """
    from bible_database import format_reference_for_speech, translate_verse_to_hindi, translate_any_text_to_hindi

    cfg = load_active_voice()
    en_voice = cfg.get("voice", "en-US-ChristopherNeural")
    if "hi-IN" in en_voice and listen_mode == "english":
        en_voice = "en-IN-PrabhatNeural"
    rate_str = cfg.get("rate", "-3%")

    out_mp3 = PREVIEWS_DIR / f"verse_{listen_mode}_{key_hash}.mp3"
